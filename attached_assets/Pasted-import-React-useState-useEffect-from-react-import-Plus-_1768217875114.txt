…games,

…acc,
[name]: { touchdowns: 0, extraPoints: 0, qbTouchdowns: 0, flagPulls: 0 }

}
});

const updateScore = (gameId, team, value) => {
setGames({
…games,
[gameId]: {
…games[gameId],
[team]: Math.max(0, games[gameId][team] + value)
}
});
};

const updateStat = (gameId, player, stat, value) => {

…games,
[gameId]: {
…games[gameId],

…games[gameId].stats,
[player]: {
…games[gameId].stats[player],
[stat]: Math.max(0, games[gameId].stats[player][stat] + value)
}
}
}
});
};
